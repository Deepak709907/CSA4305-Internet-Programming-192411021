Question 1: XML Parsing Algorithm

Pseudocode/implementation to parse an XML document containing product details, traverse its nodes, and display the extracted data.

Code

import javax.xml.parsers.*;
import org.w3c.dom.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws Exception {
        String xml = """
            <products>
                <product>
                    <id>101</id>
                    <name>Laptop</name>
                    <price>55000</price>
                </product>
                <product>
                    <id>102</id>
                    <name>Phone</name>
                    <price>30000</price>
                </product>
            </products>
            """;

        DocumentBuilderFactory f = DocumentBuilderFactory.newInstance();
        DocumentBuilder b = f.newDocumentBuilder();
        Document doc = b.parse(
            new ByteArrayInputStream(xml.getBytes())
        );

        NodeList products = doc.getElementsByTagName("product");
        System.out.println("Product Details");

        for (int i = 0; i < products.getLength(); i++) {
            Element p = (Element) products.item(i);
            String id    = p.getElementsByTagName("id").item(0).getTextContent();
            String name  = p.getElementsByTagName("name").item(0).getTextContent();
            String price = p.getElementsByTagName("price").item(0).getTextContent();

            System.out.println("ID: " + id);
            System.out.println("Name: " + name);
            System.out.println("Price: " + price);
            System.out.println();
        }
    }
}

Output

Product Details ID: 101 Name: Laptop Price: 55000 ID: 102 Name: Phone Price: 30000 === Code Execution Successful ===
